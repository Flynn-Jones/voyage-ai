    -H 'Content-Type: application/json' -d '{"destination":"Tokyo","budget_cap":20}'

{
  "result": {
    "run_id": "run-18199a48477d",
    "state": "awaiting_human_review",
    "plan": {"steps": [{"tool": "search_accommodations",
                         "reason": "To find accommodations within the $20.00/night budget in Tokyo"}],
             "rationale": "Using the search_accommodations tool will directly provide the required information on accommodations within the specified budget."},
    "work": {
      "items": [],
      "summary": "No accommodation found in Tokyo within $20.00/night. Nothing was shortlisted.",

# The Reviewer's findings/recommendations exactly match the deterministic fallback_review()
# text (validate_review grounds every run in the computed checks) even though this went through
# the real LLM path — proof the grounding logic holds on a live model, not just the mocked tests.

$ curl -s -X POST http://localhost:5002/accommodation/multi-agent/runs/run-18199a48477d/review \
    -H 'Content-Type: application/json' \
    -d '{"decision":"reject","note":"too far over budget, will widen cap instead","reviewer":"demo-user"}'

{
  "result": {
    "run_id": "run-18199a48477d",
    "decision": "reject",
    "state": "rejected",
    "approved_items": [],
    "rejected_ids": [],
    "applied_recorded": null
  },
  "status": "success"
}

# A rejected run is never reported to /applied (applied_recorded stays null) — nothing is
# left pending, and the run's terminal state is "rejected", not "applied".
